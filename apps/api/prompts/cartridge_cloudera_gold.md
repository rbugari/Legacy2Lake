## 5A. DIMENSION TABLE PATTERN (Iceberg)

```python
from pyspark.sql.functions import broadcast, monotonically_increasing_id

# Add surrogate key (hash-based for idempotency — prefer over monotonically_increasing_id)
from pyspark.sql.functions import sha2, concat_ws

df_dim = df_silver.withColumn(
    surrogate_key_col,
    sha2(concat_ws("||", *[F.col(k) for k in business_keys]), 256).cast(LongType())
).withColumn(
    "effective_date", F.current_date()
).withColumn(
    "expiration_date", F.lit(None).cast(DateType())
).withColumn(
    "is_active", F.lit(True)
).withColumn(
    "dimension_created_at", F.current_timestamp()
).withColumn(
    "dimension_updated_at", F.current_timestamp()
)

# Select business-friendly columns
dimension_cols = config.get('dimension_columns', [c for c in df_dim.columns])
df_gold_dim = df_dim.select(*dimension_cols)
```

**Dimension MERGE (SCD Type 2 on Iceberg Gold)**
```python
gold_table = f"{catalog}.{gold_schema}.dim_{config['entity_name']}"

# Create table if not exists
spark.sql(f"""
    CREATE TABLE IF NOT EXISTS {gold_table}
    USING iceberg
    TBLPROPERTIES (
        'write.merge.mode' = 'merge-on-read',
        'write.upsert.enabled' = 'true'
    )
""")

# Build conditions dynamically
merge_on = " AND ".join([f"t.{k} = s.{k}" for k in business_keys])
if not change_detection_cols:
    sys_cols = {surrogate_key_col, "effective_date", "expiration_date", "is_active",
                "dimension_created_at", "dimension_updated_at"}
    change_detection_cols = [c for c in df_gold_dim.columns if c not in set(business_keys) | sys_cols]
change_expr = " OR ".join([f"t.{c} <> s.{c}" for c in change_detection_cols])

df_gold_dim.createOrReplaceTempView("_gold_dim_incoming")

# Step 1: Expire changed records
spark.sql(f"""
    MERGE INTO {gold_table} t
    USING _gold_dim_incoming s
    ON {merge_on} AND t.is_active = true
    WHEN MATCHED AND ({change_expr}) THEN UPDATE SET
        t.expiration_date      = current_date(),
        t.is_active            = false,
        t.dimension_updated_at = current_timestamp()
""")

# Step 2: Insert new versions
spark.sql(f"INSERT INTO {gold_table} SELECT * FROM _gold_dim_incoming")
logger.info(f"[LOAD] Dimension SCD Type 2 MERGE completed: {gold_table}")
```

## 5B. FACT TABLE PATTERN (Iceberg)

```python
# Join with dimension tables to resolve surrogate keys
df_dim_customer = spark.table(f"{catalog}.{gold_schema}.dim_customer").filter("is_active = true")
df_dim_product  = spark.table(f"{catalog}.{gold_schema}.dim_product").filter("is_active = true")

df_fact = df_silver.join(
    broadcast(df_dim_customer.select("customer_id", surrogate_key_col)),
    "customer_id",
    "left"
).join(
    broadcast(df_dim_product.select("product_id", "product_sk")),
    "product_id",
    "left"
)

# Validate referential integrity
null_sk = df_fact.filter(F.col(surrogate_key_col).isNull()).count()
if null_sk > 0:
    logger.warning(f"[INTEGRITY] {null_sk} fact rows with NULL surrogate key (orphaned)")

# Calculate business measures from config parameters
df_fact = df_fact.withColumn(
    "gross_amount", F.col("quantity") * F.col("unit_price")
).withColumn(
    "discount_amount", F.col("gross_amount") * F.lit(discount_rate)
).withColumn(
    "taxable_amount", F.col("gross_amount") - F.col("discount_amount")
).withColumn(
    "tax_amount", F.col("taxable_amount") * F.lit(tax_rate)
).withColumn(
    "net_amount", F.col("taxable_amount") + F.col("tax_amount")
).withColumn(
    "fiscal_year", F.year(F.col("order_date"))
).withColumn(
    "fiscal_quarter", F.quarter(F.col("order_date"))
).withColumn(
    "fiscal_month", F.month(F.col("order_date"))
).withColumn(
    "fact_created_at", F.current_timestamp()
)

# Fact MERGE (dedup-on-grain)
fact_table = f"{catalog}.{gold_schema}.fact_{config['entity_name']}"

spark.sql(f"""
    CREATE TABLE IF NOT EXISTS {fact_table}
    USING iceberg
    TBLPROPERTIES ('write.merge.mode' = 'copy-on-write')
""")

grain_keys = config.get('grain_keys', ['order_id'])
merge_on_fact = " AND ".join([f"t.{k} = s.{k}" for k in grain_keys])
df_fact.createOrReplaceTempView("_gold_fact_incoming")

spark.sql(f"""
    MERGE INTO {fact_table} t
    USING _gold_fact_incoming s
    ON {merge_on_fact}
    WHEN MATCHED THEN UPDATE SET *
    WHEN NOT MATCHED THEN INSERT *
""")
logger.info(f"[LOAD] Fact MERGE completed: {fact_table}")
```

## 5C. AGGREGATE / METRIC TABLE PATTERN (Iceberg)

```python
if not aggregation_grain:
    raise ValueError("config['aggregation_grain'] must be provided for AGGREGATE table type")

df_agg = df_silver.groupBy(*aggregation_grain).agg(
    F.count("*").alias("record_count"),
    F.sum(config.get('measure_col', 'amount')).alias("total_amount"),
    F.avg(config.get('measure_col', 'amount')).alias("avg_amount"),
    F.min(config.get('date_col', 'load_date')).alias("first_date"),
    F.max(config.get('date_col', 'load_date')).alias("last_date")
)

# Add revenue ranking window
from pyspark.sql.window import Window
rank_window = Window.partitionBy(
    [g for g in aggregation_grain if "year" in g or "period" in g]
).orderBy(F.desc("total_amount"))

df_agg = df_agg.withColumn(
    "revenue_rank", F.rank().over(rank_window)
).withColumn(
    "revenue_percentile", F.percent_rank().over(rank_window)
).withColumn(
    "aggregate_created_at", F.current_timestamp()
)

# Full refresh via Iceberg overwrite (aggregates are recalculated)
agg_table = f"{catalog}.{gold_schema}.agg_{config['metric_name']}"
df_agg.writeTo(agg_table).using("iceberg").createOrReplace()
logger.info(f"[LOAD] Aggregate table refreshed: {agg_table}")
```

### 8. IMPALA COMPATIBILITY TIPS

- Use `DateType` for date keys — Impala handles DATE efficiently.
- Gold table names should follow `dim_`, `fact_`, `agg_` prefixes for Impala catalog discoverability.
