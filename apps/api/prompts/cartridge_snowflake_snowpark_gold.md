### 2. SNOWFLAKE GOLD LAYER FUNDAMENTALS

**⚠️ CRITICAL: Function signature is `execute_task(session, config)` — Snowpark `Session`, NOT PySpark `SparkSession`.**

Gold layer is the analytical presentation layer. It has **3 structural patterns**:
- **DIMENSION**: Hash-based surrogate keys (SHA2), SCD Type 2 via `MERGE INTO`, one active record per business key.
- **FACT**: Resolves business keys to surrogate keys (dimension lookups), computes business measures, append or full-overwrite.
- **AGGREGATE**: `groupBy().agg()` with window functions, overwrite each run, feeds BI layer directly.

**Snowflake-specific Gold advantages:**
- Zero-copy clones for dev/test environments (orchestrator manages, no code changes)
- Dynamic Tables for real-time aggregate materialization (optional, config-driven)
- Automatic clustering within Snowflake (no explicit shuffle/repartition)
- Tight Power BI / Tableau / Sigma integration via Snowflake Connector

**Auto-detection from Agent C's architectural decision:**
```python
model_type = config.get('model_type', 'DIMENSION').upper()
# → 'DIMENSION' | 'FACT' | 'AGGREGATE'
```

### 4. SURROGATE KEY GENERATION (Dimension Pattern)
```python
from snowflake.snowpark import functions as F

def build_surrogate_key(df, business_keys: list, sk_col: str):
    """
    Generate a deterministic SHA2-based surrogate key.
    Snowflake-native: F.sha2(F.concat_ws(F.lit("||"), *cols), F.lit(256))
    """
    concatenated = F.concat_ws(
        F.lit("||"),
        *[F.coalesce(F.col(k).cast("string"), F.lit("NULL")) for k in business_keys]
    )
    return df.with_column(sk_col, F.sha2(concatenated, F.lit(256)))
```

### 5. DIMENSION PATTERN (Full SCD Type 2)
```python
### --- DIMENSION PATTERN START ---

silver_full = f"{database}.{silver_schema}.{source_table}"
gold_full   = f"{database}.{gold_schema}.{target_table}"

df_silver = session.table(silver_full).filter(F.col("IS_CURRENT") == F.lit(True))
source_count = df_silver.count()
logger.info(f"[EXTRACT] {source_count} active Silver rows")

# 1. Build surrogate key
sk_col = config.get('sk_column', f"{target_table}_SK")
df_sk = build_surrogate_key(df_silver, business_keys, sk_col)

# 2. Add SCD Type 2 Gold columns
df_gold_incoming = df_sk \
    .with_column("VALID_FROM",   F.col("VALID_FROM") if "VALID_FROM" in [c.name for c in df_sk.schema.fields] else F.current_timestamp()) \
    .with_column("VALID_UNTIL",  F.lit(None).cast("timestamp")) \
    .with_column("IS_CURRENT",   F.lit(True)) \
    .with_column("GOLD_CREATED_AT", F.current_timestamp()) \
    .with_column("GOLD_UPDATED_AT", F.current_timestamp())

# 3. Create table if not exists
try:
    session.table(gold_full).count()
    table_exists = True
except Exception:
    table_exists = False
    df_gold_incoming.write.mode("overwrite").save_as_table(gold_full)
    logger.info(f"[INIT] Created Gold DIMENSION table: {gold_full}")

if table_exists:
    df_gold_incoming.create_or_replace_temp_view("DIM_INCOMING")

    merge_on     = " AND ".join([f"t.{k} = s.{k}" for k in business_keys])
    # Dynamic change detection
    if not change_cols:
        reserved = {sk_col, "VALID_FROM", "VALID_UNTIL", "IS_CURRENT",
                    "GOLD_CREATED_AT", "GOLD_UPDATED_AT"}
        change_cols = [c.name for c in df_gold_incoming.schema.fields
                       if c.name not in set(business_keys) | reserved]

    change_expr  = " OR ".join([f"t.{c} <> s.{c}" for c in change_cols])

    # Step 1: Expire changed records
    session.sql(f"""
        MERGE INTO {gold_full} t
        USING DIM_INCOMING s
        ON {merge_on} AND t.IS_CURRENT = true
        WHEN MATCHED AND ({change_expr}) THEN UPDATE SET
            t.VALID_UNTIL       = current_timestamp(),
            t.IS_CURRENT        = false,
            t.GOLD_UPDATED_AT   = current_timestamp()
    """).collect()

    # Step 2: Insert new versions
    all_cols = ", ".join([c.name for c in df_gold_incoming.schema.fields])
    session.sql(f"""
        INSERT INTO {gold_full} ({all_cols})
        SELECT {all_cols} FROM DIM_INCOMING
    """).collect()

    logger.info(f"[LOAD] DIMENSION SCD Type 2 MERGE completed: {gold_full}")

active_count = session.sql(f"SELECT COUNT(*) AS CNT FROM {gold_full} WHERE IS_CURRENT = true").collect()[0]["CNT"]
logger.info(f"[LOAD] {active_count} active dimension records")

### --- DIMENSION PATTERN END ---
```

### 6. FACT PATTERN (SK Resolution + Append)
```python
### --- FACT PATTERN START ---

silver_full = f"{database}.{silver_schema}.{source_table}"
gold_full   = f"{database}.{gold_schema}.{target_table}"

df_fact_src = session.table(silver_full).filter(F.col("IS_CURRENT") == F.lit(True))
source_count = df_fact_src.count()
logger.info(f"[EXTRACT] {source_count} rows for Fact")

# Surrogate Key resolution — join each dimension
df_fact = df_fact_src
for dim_cfg in dimension_tables:
    dim_full     = f"{database}.{gold_schema}.{dim_cfg['table']}"
    source_bk    = dim_cfg['source_bk']    # business key in the fact source
    dim_bk       = dim_cfg['dim_bk']       # business key in the dimension
    dim_sk_col   = dim_cfg['sk_col']       # surrogate key column name in dim
    alias_prefix = dim_cfg.get('prefix', dim_cfg['table'].upper())

    df_dim = session.table(dim_full) \
        .filter(F.col("IS_CURRENT") == F.lit(True)) \
        .select(dim_bk, dim_sk_col)

    df_fact = df_fact.join(
        df_dim,
        df_fact[source_bk] == df_dim[dim_bk],
        join_type="left"
    ).with_column(f"{alias_prefix}_SK", F.col(dim_sk_col)) \
     .drop(dim_bk, dim_sk_col)

    logger.info(f"[TRANSFORM] Resolved SK from dimension: {dim_cfg['table']}")

# Compute business measures
for m in fact_measures:
    col    = m.get('col')
    func   = m.get('func', 'sum').upper()
    alias  = m.get('alias', col)
    dtype  = m.get('dtype', 'DECIMAL(18,4)')

    if func == 'ROUND':
        df_fact = df_fact.with_column(alias, F.round(F.col(col), m.get('scale', 2)))
    elif func == 'COALESCE_ZERO':
        df_fact = df_fact.with_column(alias, F.coalesce(F.col(col), F.lit(0)).cast(dtype))
    else:
        df_fact = df_fact.with_column(alias, F.col(col))

# Add Gold audit columns
df_fact = df_fact \
    .with_column("GOLD_CREATED_AT", F.current_timestamp()) \
    .with_column("FACT_PERIOD",     F.current_date())

# Append to Fact table (full overwrite only if config says so)
fact_mode = config.get('fact_write_mode', 'append')

if fact_mode == 'overwrite':
    df_fact.write.mode("overwrite").save_as_table(gold_full)
    logger.info(f"[LOAD] Fact OVERWRITE: {gold_full}")
else:
    df_fact.write.mode("append").save_as_table(gold_full)
    logger.info(f"[LOAD] Fact APPEND: {gold_full}")

fact_total = session.table(gold_full).count()
logger.info(f"[LOAD] Total fact rows: {fact_total}")

### --- FACT PATTERN END ---
```

### 7. AGGREGATE PATTERN (GroupBy → Full Overwrite)
```python
### --- AGGREGATE PATTERN START ---

silver_full = f"{database}.{silver_schema}.{source_table}"
gold_full   = f"{database}.{gold_schema}.{target_table}"

df_agg_src = session.table(silver_full).filter(
    F.col("IS_CURRENT") == F.lit(True)
    if "IS_CURRENT" in [c.name for c in session.table(silver_full).schema.fields]
    else F.lit(True)
)

source_count = df_agg_src.count()
logger.info(f"[EXTRACT] {source_count} source rows for aggregation")

# Build aggregation from config
from snowflake.snowpark import functions as F

agg_map = {
    "SUM":               lambda c: F.sum(F.col(c)),
    "COUNT":             lambda c: F.count(F.col(c)),
    "COUNT_DISTINCT":    lambda c: F.count_distinct(F.col(c)),
    "AVG":               lambda c: F.avg(F.col(c)),
    "MAX":               lambda c: F.max(F.col(c)),
    "MIN":               lambda c: F.min(F.col(c)),
}

agg_exprs = []
for a in agg_definitions:
    fn  = agg_map.get(a.get('func', 'SUM').upper(), F.sum)
    agg_exprs.append(fn(a['col']).alias(a.get('alias', a['col'])))

df_aggregated = df_agg_src.group_by(*group_by_cols).agg(*agg_exprs)

# Window-based measures (running totals, rankings, percentages)
for w_def in config.get('window_measures', []):
    partition_cols = w_def.get('partition_by', group_by_cols)
    order_cols     = w_def.get('order_by', [])
    func           = w_def.get('func', 'RANK')
    alias          = w_def.get('alias', f"WIN_{func}")

    from snowflake.snowpark.window import Window as WF
    win_spec = WF.partition_by(*partition_cols)
    if order_cols:
        win_spec = win_spec.order_by(*[F.col(c) for c in order_cols])

    if func == "RANK":
        df_aggregated = df_aggregated.with_column(alias, F.rank().over(win_spec))
    elif func == "ROW_NUMBER":
        df_aggregated = df_aggregated.with_column(alias, F.row_number().over(win_spec))
    elif func == "CUMSUM":
        measure_col = w_def.get('measure_col')
        df_aggregated = df_aggregated.with_column(alias, F.sum(F.col(measure_col)).over(win_spec))

# Add metadata
df_aggregated = df_aggregated \
    .with_column("GOLD_CREATED_AT", F.current_timestamp()) \
    .with_column("AGGREGATE_PERIOD", F.current_date())

# Full overwrite per run (aggregates always replaced)
df_aggregated.write.mode("overwrite").save_as_table(gold_full)
logger.info(f"[LOAD] Aggregate OVERWRITE: {gold_full}")

agg_count = df_aggregated.count()
logger.info(f"[LOAD] {agg_count} aggregate rows written")

### --- AGGREGATE PATTERN END ---
```
