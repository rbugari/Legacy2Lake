### 2. BIGQUERY SQL PLATFORM NOTES

**⚠️ CRITICAL: Pure BigQuery SQL Scripting — no Spark, no Python, no `session` object.**

- **Naming**: `` `PROJECT.DATASET.TABLE` `` — backtick-quoted in all DML.
- **Deduplication**: BigQuery's `QUALIFY ROW_NUMBER() OVER (...) = 1` — exact same syntax as Snowflake; Snowflake borrowed it from BigQuery.
- **MERGE**: Full `MERGE dataset.target T USING (...) S ON ... WHEN MATCHED ... WHEN NOT MATCHED ...`.
- **TEMP TABLEs**: Session-scoped, no storage cost, auto-dropped at end of script.
- **`APPROX_COUNT_DISTINCT`**: Preferred over `COUNT(DISTINCT)` for quality checks on large tables (cost + performance).
- **Partitioning & Clustering**: Defined at table DDL level. No per-run shuffle needed.
- **`@@row_count`**: System variable available after any DML statement.
- **`INFORMATION_SCHEMA`**: Used for table existence checks and schema introspection.

### 5. DEDUPLICATION (BigQuery QUALIFY Idiom)

**`QUALIFY` eliminates duplicates without a subquery — same pattern as Snowflake:**

```sql
-- Step 1: Deduplicate Bronze — QUALIFY is native BigQuery SQL
CREATE TEMP TABLE SILVER_DEDUP AS
SELECT * EXCEPT (_ingestion_timestamp, _ingestion_date, _source_file, _record_hash),
       _ingestion_timestamp AS _bronze_ts
FROM (
    SELECT *,
        ROW_NUMBER() OVER (
            PARTITION BY BUSINESS_KEY_1, BUSINESS_KEY_2   -- from p_business_keys
            ORDER BY _ingestion_timestamp DESC
        ) AS _rn
    FROM `project.bronze_dataset.source_table`   -- via EXECUTE IMMEDIATE with v_bronze_full
    WHERE _ingestion_date >= p_batch_date
)
WHERE _rn = 1;   -- BigQuery supports QUALIFY inline or as a subquery filter
```

**Dynamic version using `EXECUTE IMMEDIATE`:**
```sql
EXECUTE IMMEDIATE FORMAT("""
    CREATE TEMP TABLE SILVER_DEDUP AS
    SELECT * EXCEPT (_ingestion_timestamp, _ingestion_date, _source_file, _record_hash,  _rn),
           _ingestion_timestamp AS _bronze_ts
    FROM (
        SELECT *,
            ROW_NUMBER() OVER (
                PARTITION BY %s
                ORDER BY _ingestion_timestamp DESC
            ) AS _rn
        FROM %s
        WHERE _ingestion_date >= DATE('%s')
    )
    WHERE _rn = 1
""",
    -- Replace p_business_keys 'BK1,BK2' → 'BK1, BK2'
    REPLACE(p_business_keys, ',', ', '),
    v_bronze_full,
    CAST(p_batch_date AS STRING)
);
```

### 6. DATA QUALITY (TEMP TABLE, Non-Blocking)

```sql
-- Quality flags — non-blocking, all records proceed to Silver
CREATE TEMP TABLE SILVER_QUALITY AS
SELECT
    *,
    (BUSINESS_KEY_1 IS NOT NULL AND BUSINESS_KEY_2 IS NOT NULL)  AS IS_COMPLETE_RECORD,
    (AMOUNT >= 0 OR AMOUNT IS NULL)                               AS IS_AMOUNT_VALID,
    (LOAD_DATE IS NOT NULL)                                       AS IS_DATE_VALID
FROM SILVER_DEDUP;

-- Log quality stats (non-blocking)
DECLARE v_incomplete INT64;
SET v_incomplete = (SELECT COUNTIF(NOT IS_COMPLETE_RECORD) FROM SILVER_QUALITY);

IF v_incomplete > 0 THEN
    INSERT INTO {{dataset}}.L2L_QUALITY_LOG
        (ASSET_NAME, LAYER, BATCH_DATE, RULE, FAILING_ROWS, EXECUTED_AT)
    VALUES
        ('{{asset_name}}', 'SILVER', p_batch_date, 'NULL_BUSINESS_KEY', v_incomplete, CURRENT_TIMESTAMP());
END IF;
```

### 7. SCD TYPE 2 — BIGQUERY MERGE (Two-Step Pattern)

**Step 1 — Create Silver table if not exists:**
```sql
DECLARE v_table_exists BOOL DEFAULT (
    SELECT COUNT(*) > 0
    FROM `region-us`.INFORMATION_SCHEMA.TABLES
    WHERE table_catalog = p_project
      AND table_schema  = p_silver_dataset
      AND table_name    = p_target_table
);

IF NOT v_table_exists THEN
    EXECUTE IMMEDIATE FORMAT("""
        CREATE TABLE %s (
            BUSINESS_KEY_1      STRING    NOT NULL,
            BUSINESS_KEY_2      STRING,
            ATTRIBUTE_A         STRING,
            AMOUNT              NUMERIC,
            LOAD_DATE           DATE,
            VALID_FROM          TIMESTAMP NOT NULL,
            VALID_UNTIL         TIMESTAMP,
            IS_CURRENT          BOOL      NOT NULL,
            RECORD_VERSION      INT64     NOT NULL,
            IS_COMPLETE_RECORD  BOOL,
            SILVER_CREATED_AT   TIMESTAMP,
            SILVER_UPDATED_AT   TIMESTAMP
        )
        PARTITION BY DATE(VALID_FROM)
        CLUSTER BY BUSINESS_KEY_1, IS_CURRENT
        OPTIONS (description = 'L2L Silver SCD2: {{asset_name}}')
    """, v_silver_full);
END IF;
```

**Step 2 — SCD Type 2 MERGE (two-step):**
```sql
IF p_scd_enabled THEN

    -- STEP A: Expire changed active records
    EXECUTE IMMEDIATE FORMAT("""
        MERGE %s T
        USING (SELECT * FROM SILVER_QUALITY) S
        ON T.BUSINESS_KEY_1 = S.BUSINESS_KEY_1
           AND T.BUSINESS_KEY_2 = S.BUSINESS_KEY_2
           AND T.IS_CURRENT = TRUE
        WHEN MATCHED AND (
            T.ATTRIBUTE_A <> S.ATTRIBUTE_A OR
            T.AMOUNT      <> S.AMOUNT
        ) THEN UPDATE SET
            T.VALID_UNTIL       = CURRENT_TIMESTAMP(),
            T.IS_CURRENT        = FALSE,
            T.SILVER_UPDATED_AT = CURRENT_TIMESTAMP()
    """, v_silver_full);

    -- STEP B: Insert all incoming as new versions
    EXECUTE IMMEDIATE FORMAT("""
        INSERT INTO %s
            (BUSINESS_KEY_1, BUSINESS_KEY_2, ATTRIBUTE_A, AMOUNT, LOAD_DATE,
             VALID_FROM, VALID_UNTIL, IS_CURRENT, RECORD_VERSION,
             IS_COMPLETE_RECORD, SILVER_CREATED_AT, SILVER_UPDATED_AT)
        SELECT
            BUSINESS_KEY_1, BUSINESS_KEY_2, ATTRIBUTE_A, AMOUNT, LOAD_DATE,
            CURRENT_TIMESTAMP(),   -- VALID_FROM
            NULL,                  -- VALID_UNTIL
            TRUE,                  -- IS_CURRENT
            COALESCE(
                (SELECT MAX(t2.RECORD_VERSION) + 1
                 FROM %s t2
                 WHERE t2.BUSINESS_KEY_1 = s.BUSINESS_KEY_1
                   AND t2.BUSINESS_KEY_2 = s.BUSINESS_KEY_2), 1
            ),                     -- RECORD_VERSION
            IS_COMPLETE_RECORD,
            CURRENT_TIMESTAMP(),   -- SILVER_CREATED_AT
            CURRENT_TIMESTAMP()    -- SILVER_UPDATED_AT
        FROM SILVER_QUALITY s
    """, v_silver_full, v_silver_full);

    SET v_rows_merged = @@row_count;

ELSE
    -- Simple upsert (SCD disabled)
    EXECUTE IMMEDIATE FORMAT("""
        MERGE %s T
        USING (SELECT * FROM SILVER_QUALITY) S
        ON T.BUSINESS_KEY_1 = S.BUSINESS_KEY_1
           AND T.BUSINESS_KEY_2 = S.BUSINESS_KEY_2
        WHEN MATCHED THEN UPDATE SET
            T.ATTRIBUTE_A       = S.ATTRIBUTE_A,
            T.AMOUNT            = S.AMOUNT,
            T.SILVER_UPDATED_AT = CURRENT_TIMESTAMP()
        WHEN NOT MATCHED THEN INSERT ROW
    """, v_silver_full);

    SET v_rows_merged = @@row_count;

END IF;
```

### 9. BIGQUERY NATIVE OPTIMIZATION HINTS

**Clustering Column Order — BI Engine Acceleration:**
```sql
-- Column ORDER in CLUSTER BY matters. Most selective equality filter comes FIRST.
-- For Silver with IS_CURRENT pattern:
CLUSTER BY IS_CURRENT, BUSINESS_KEY_1
-- ✅ Best for: WHERE IS_CURRENT=TRUE AND BUSINESS_KEY_1='X'  (IS_CURRENT is binary = ultra-selective)
-- ❌ NOT: CLUSTER BY BUSINESS_KEY_1, IS_CURRENT  (high-cardinality first = reduces IS_CURRENT pruning)
```

**Partition Pruning — Use DATE() in WHERE:**
```sql
-- BQ prunes partitions only when the WHERE clause matches the PARTITION BY expression:
-- VALID:   WHERE DATE(VALID_FROM) >= '2026-01-01'  → partition pruning ✅
-- INVALID: WHERE VALID_FROM >= '2026-01-01'        → NO pruning if PARTITION BY DATE(VALID_FROM)
-- RULE: ALWAYS use DATE(timestamp_col) in WHERE to match the partition expression exactly.
```

**Materialized View for IS_CURRENT=TRUE (High-Read Silver):**
```sql
-- If Silver is queried many times per hour by Gold procedures, pre-filter with MV:
CREATE MATERIALIZED VIEW `{{project}}.{{silver_dataset}}.MV_{{target_table}}_CURRENT`
OPTIONS (
    enable_refresh                    = true,
    refresh_interval_minutes          = 30,
    allow_non_incremental_definition  = true
)
AS
SELECT
    BUSINESS_KEY_1, BUSINESS_KEY_2, ATTRIBUTE_A, AMOUNT,
    VALID_FROM, RECORD_VERSION
FROM `{{project}}.{{silver_dataset}}.{{target_table}}`
WHERE IS_CURRENT = TRUE;
-- BQ auto-refreshes MV every 30 min. Gold SP queries MV instead of the full Silver table.
```

**@@script.bytes_processed — Cost Tracking:**
```sql
DECLARE v_bytes INT64;
SET v_bytes = @@script.bytes_processed;

INSERT INTO {{dataset}}.L2L_EXECUTION_LOG
    (ASSET_NAME, LAYER, BATCH_DATE, ROWS_LOADED, BYTES_PROCESSED, STATUS, EXECUTED_AT)
VALUES
    ('{{asset_name}}', 'SILVER', p_batch_date, v_rows_merged, v_bytes, 'SUCCESS', CURRENT_TIMESTAMP());
```

**APPROX_COUNT_DISTINCT — Quality Checks on Large Tables:**
```sql
-- For Silver tables > 10M rows, ALWAYS prefer APPROX over exact COUNT(DISTINCT):
-- APPROX_COUNT_DISTINCT(col)  → ~1-2% error, 10–100x cheaper in bytes scanned
-- COUNT(DISTINCT col)         → exact but full table scan (expensive at scale)
-- Use COUNT(DISTINCT) only when exact cardinality is required for a hard business rule.
```

**Column-Level Security for PII (run once as DDL, outside SP):**
```sql
-- Option A: BigQuery column data policies (requires Policy Tags via Data Catalog UI/CLI)
-- Apply policy tag to PII columns via BQ CLI:
-- bq update --schema ... (column policy tags are API/UI-only, not available in SQL scripting)

-- Option B: Hash PII in Silver procedure before writing:
TO_HEX(SHA256(CAST(email_col AS BYTES)))  AS EMAIL_HASHED  -- deterministic, reversal-resistant
-- Option C: Create a Silver MASKED VIEW for analyst access:
CREATE OR REPLACE VIEW `{{project}}.{{silver_dataset}}.VW_{{target_table}}_MASKED` AS
SELECT
    BUSINESS_KEY_1,
    TO_HEX(SHA256(CAST(EMAIL AS BYTES)))    AS EMAIL_HASHED,
    TO_HEX(SHA256(CAST(PHONE AS BYTES)))    AS PHONE_HASHED,
    AMOUNT, VALID_FROM, IS_CURRENT
FROM `{{project}}.{{silver_dataset}}.{{target_table}}`;
```
