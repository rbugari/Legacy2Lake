## 📋 EXAMPLE TEMPLATE STRUCTURE
```python
# ================================================================
# L2L DIRECT TRANSLATION: DimCustomer.dtsx
# ================================================================
# Source Asset: DimCustomer.dtsx
# Source Technology: Microsoft SSIS
# Target Platform: Microsoft Fabric (Lakehouse + PySpark + Delta Lake)
# Translation Mode: Direct 1:1 (No Architectural Patterns)
# Component Type: Data Pipeline
# Business Entity: Customer Dimension
# Load Strategy: FULL_OVERWRITE
# Generated At: 2026-02-16T12:00:00Z
# Generated By: L2L Agent-C (Cartridge: DirectTranspiler v4.0 - Fabric)
# ================================================================
# NOTE: This is a functional translation that preserves original logic.
#       For architectural patterns (Medallion, Data Vault, etc.), 
#       apply Pattern Application phase after validation.
# ================================================================

from pyspark.sql import functions as F
from pyspark.sql.types import *
import logging

def execute_task(spark, config):
    """
    Direct translation of DimCustomer.dtsx to Microsoft Fabric PySpark.
    
    Preserves original SSIS logic:
    - OLE DB Source → JDBC read
    - Type conversions → explicit casting
    - Destination → Delta Lake write (overwrite)
    
    Config parameters (required):
    - jdbc_url: Source database connection
    - jdbc_user, jdbc_password: Credentials
    - source_query: SQL query from SSIS source
    - target_path: Fabric lakehouse path
    - columns: Column metadata from Sprint 10
    - load_mode: 'overwrite' for FULL_OVERWRITE strategy
    """
    logger = logging.getLogger(__name__)
    
    # ============================================================
    # [EXTRACT] Read from source using SSIS query
    # ============================================================
    try:
        source_query = config['source_query']
        query_with_alias = f"({source_query}) AS source"
        
        logger.info(f"[EXTRACT] Reading from source database")
        df_source = spark.read.jdbc(
            url=config['jdbc_url'],
            table=query_with_alias,
            properties={
                "user": config['jdbc_user'],
                "password": config['jdbc_password']
            }
        )
        
        source_count = df_source.count()
        logger.info(f"[EXTRACT] Source rows: {source_count}")
    except Exception as e:
        logger.error(f"[EXTRACT ERROR] {str(e)}")
        return {'status': 'failed', 'error': str(e), 'stage': 'extract'}
    
    # ============================================================
    # [TRANSFORM] Apply type casting from metadata
    # ============================================================
    try:
        columns_metadata = config.get('columns', [])
        
        logger.info(f"[TRANSFORM] Casting {len(columns_metadata)} columns")
        df_typed = df_source.select([
            F.col(col['name']).cast(_parse_data_type(col['data_type'])).alias(col['name'])
            for col in columns_metadata
        ])
        
        transform_count = df_typed.count()
        logger.info(f"[TRANSFORM] Transformed rows: {transform_count}")
    except Exception as e:
        logger.error(f"[TRANSFORM ERROR] {str(e)}")
        return {'status': 'failed', 'error': str(e), 'stage': 'transform'}
    
    # ============================================================
    # [LOAD] Write to Fabric Lakehouse (Simple Overwrite)
    # ============================================================
    try:
        target_path = config['target_path']
        load_mode = config.get('load_mode', 'overwrite')
        
        logger.info(f"[LOAD] Writing to {target_path} (mode={load_mode})")
        df_typed.write.format("delta").mode(load_mode).save(target_path)
        
        logger.info(f"[LOAD] Loaded {transform_count} rows to {target_path}")
    except Exception as e:
        logger.error(f"[LOAD ERROR] {str(e)}")
        return {'status': 'failed', 'error': str(e), 'stage': 'load'}
    
    # ============================================================
    # [RETURN] Success status
    # ============================================================
    return {
        'status': 'success',
        'rows_written': transform_count,
        'target_path': target_path
    }

def _parse_data_type(dtype_str):
    """Convert metadata string to PySpark type"""
    dtype_map = {
        'STRING': StringType(),
        'INT': IntegerType(),
        'BIGINT': LongType(),
        'DOUBLE': DoubleType(),
        'DECIMAL': DecimalType(38, 10),
        'DATE': DateType(),
        'TIMESTAMP': TimestampType(),
        'BOOLEAN': BooleanType()
    }
    return dtype_map.get(str(dtype_str).upper(), StringType())
```
