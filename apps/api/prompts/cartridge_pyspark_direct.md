### 6. SIMPLE LOAD PATTERN (NO Medallion Architecture)
```python
def _write_target(df, config, logger):
    """
    Write DataFrame to target location.
    Supports: Hive tables, Parquet files, CSV files
    """
    database = config.get('database_name')
    table_name = config['table_name']
    target_path = config.get('target_path')
    load_mode = config.get('load_mode', 'append')
    output_format = config.get('format', 'parquet')
    
    if database and table_name:
        # Write to Hive/Metastore table
        full_table_name = f"{database}.{table_name}" if database else table_name
        
        df.write \
            .format(output_format) \
            .mode(load_mode) \
            .option("path", target_path if target_path else None) \
            .saveAsTable(full_table_name)
        
        logger.info(f"✅ Saved to table: {full_table_name} (mode={load_mode})")
        return full_table_name
        
    elif target_path:
        # Write to file path
        df.write \
            .format(output_format) \
            .mode(load_mode) \
            .option("header", "true") \
            .save(target_path)
        
        logger.info(f"✅ Saved to path: {target_path} (format={output_format})")
        return target_path
    
    else:
        raise ValueError("Must provide either (database + table_name) or target_path")
```

**RATIONALE**: Direct translation uses simple write operations (append/overwrite). MERGE/UPSERT patterns are architectural enhancements applied in Refinement phase.

# L2L DIRECT TRANSLATION: CustomerOrders
# ================================================================
# Source Asset: CustomerOrders.dtsx (SSIS Package)
# Source Technology: Microsoft SQL Server (SSIS)
# Target Platform: PySpark (Generic)
# Translation Mode: Direct 1:1 (No Architectural Patterns)
# Component Type: Data Pipeline
# Business Entity: CustomerOrders
# Load Strategy: Full Load (Overwrite)
# Generated At: 2026-02-19 14:30:00 UTC
# Generated By: L2L Agent-C (Cartridge: PySpark Direct v4.0)
# ================================================================

import logging
from pyspark.sql import SparkSession
from pyspark.sql import functions as F
from pyspark.sql.types import *

def execute_task(spark, config):
    """
    Direct translation of CustomerOrders SSIS package to PySpark.
    Extracts from SQL Server, applies transformations, loads to target.
    """
    logger = logging.getLogger(__name__)
    logger.info(f"Starting CustomerOrders translation")
    
    try:
        # =============================================================
        # EXTRACT - Read from SQL Server using JDBC
        # =============================================================
        source_query = config.get('source_query', "(SELECT * FROM Sales.CustomerOrders) AS src")
        df_source = spark.read.jdbc(
            url=config['jdbc_url'],
            table=source_query,
            properties={
                "user": config['jdbc_user'],
                "password": config['jdbc_password'],
                "driver": "com.microsoft.sqlserver.jdbc.SQLServerDriver"
            }
        )
        
        source_count = df_source.count()
        logger.info(f"✅ Extracted {source_count} rows from source")
        
        # =============================================================
        # TRANSFORM - Apply business logic (preserve original SSIS logic)
        # =============================================================
        
        # Cast columns based on metadata
        columns_metadata = config.get('columns', [])
        df_typed = df_source
        for col_meta in columns_metadata:
            df_typed = df_typed.withColumn(
                col_meta['name'],
                F.col(col_meta['name']).cast(_parse_data_type(col_meta['data_type']))
            )
        
        # Original SSIS Derived Column: OrderYear = YEAR(OrderDate)
        df_transformed = df_typed.withColumn("OrderYear", F.year(F.col("OrderDate")))
        
        # Original SSIS Derived Column: FullName = FirstName + " " + LastName
        df_transformed = df_transformed.withColumn(
            "FullName",
            F.concat(F.col("FirstName"), F.lit(" "), F.col("LastName"))
        )
        
        # Original SSIS Conditional Split: HighValue = Amount > 1000
        df_transformed = df_transformed.withColumn(
            "HighValue",
            F.when(F.col("Amount") > 1000, "Y").otherwise("N")
        )
        
        logger.info(f"✅ Transformations applied")
        
        # =============================================================
        # LOAD - Write to target (Parquet or Hive table)
        # =============================================================
        database = config.get('database_name')
        table_name = config['table_name']
        target_path = config.get('target_path')
        load_mode = config.get('load_mode', 'overwrite')
        output_format = config.get('format', 'parquet')
        
        if database and table_name:
            full_table_name = f"{database}.{table_name}"
            df_transformed.write \
                .format(output_format) \
                .mode(load_mode) \
                .saveAsTable(full_table_name)
            output_location = full_table_name
        else:
            df_transformed.write \
                .format(output_format) \
                .mode(load_mode) \
                .option("header", "true") \
                .save(target_path)
            output_location = target_path
        
        final_count = df_transformed.count()
        logger.info(f"✅ Loaded {final_count} rows to {output_location}")
        
        # =============================================================
        # RETURN - Execution summary
        # =============================================================
        return {
            "status": "SUCCESS",
            "rows_extracted": source_count,
            "rows_loaded": final_count,
            "output_location": output_location,
            "load_mode": load_mode
        }
        
    except Exception as e:
        logger.error(f"❌ Execution failed: {str(e)}", exc_info=True)
        return {
            "status": "FAILED",
            "error": str(e),
            "rows_extracted": 0,
            "rows_loaded": 0
        }

def _parse_data_type(dtype_str):
    """Convert metadata string to PySpark type"""
    dtype_map = {
        'STRING': StringType(),
        'VARCHAR': StringType(),
        'INT': IntegerType(),
        'BIGINT': LongType(),
        'DOUBLE': DoubleType(),
        'DECIMAL': DecimalType(38, 10),
        'DATE': DateType(),
        'TIMESTAMP': TimestampType(),
        'BOOLEAN': BooleanType()
    }
    return dtype_map.get(dtype_str.upper(), StringType())

## 🎓 PHILOSOPHY: Drafting = "Make It Work"

- ✅ Focus on correctness and equivalence

- ✅ Use metadata for configuration

**Refinement Phase Goal:** Apply architectural patterns and optimizations.

**Separation of Concerns:**
