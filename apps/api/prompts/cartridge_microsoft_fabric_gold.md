### 6. GOLD LAYER PATTERNS BY TABLE TYPE

## 6A. DIMENSION TABLE PATTERN (Delta MERGE + SCD Type 2)

```python
from pyspark.sql.functions import sha2, concat_ws, broadcast
from delta.tables import DeltaTable

# Hash-based surrogate key — idempotent across reruns
df_dim = df_silver.withColumn(
    surrogate_key_col,
    sha2(concat_ws("||", *[F.col(k) for k in business_keys]), 256)
).withColumn(
    "effective_date", F.current_date()
).withColumn(
    "expiration_date", F.lit(None).cast("date")
).withColumn(
    "is_active", F.lit(True)
).withColumn(
    "dimension_created_at", F.current_timestamp()
).withColumn(
    "dimension_updated_at", F.current_timestamp()
)

# Restrict to business-friendly columns
dim_cols = config.get('dimension_columns', list(df_dim.columns))
df_gold_dim = df_dim.select(*dim_cols)

# Delta path for Gold Dimension
dim_path = f"{lakehouse_path}/Tables/{gold_schema}/dim_{config['entity_name']}"

if not DeltaTable.isDeltaTable(spark, dim_path):
    df_gold_dim.limit(0).write.format("delta").mode("append").save(dim_path)
    logger.info(f"[INIT] Created Gold Dimension table: {dim_path}")

dim_delta = DeltaTable.forPath(spark, dim_path)

# Change detection — auto-detect if not provided
if not change_detection_cols:
    sys_cols = {surrogate_key_col, "effective_date", "expiration_date", "is_active",
                "dimension_created_at", "dimension_updated_at"}
    change_detection_cols = [c for c in df_gold_dim.columns if c not in set(business_keys) | sys_cols]

merge_on   = " AND ".join([f"target.{k} = source.{k}" for k in business_keys])
change_cond= " OR ".join([f"target.{c} <> source.{c}" for c in change_detection_cols])

# Step 1: Expire changed records
dim_delta.alias("target").merge(
    df_gold_dim.alias("source"),
    f"{merge_on} AND target.is_active = true"
).whenMatchedUpdate(
    condition=change_cond,
    set={
        "expiration_date"     : "current_date()",
        "is_active"           : "false",
        "dimension_updated_at": "current_timestamp()"
    }
).execute()

# Step 2: Insert new / changed versions
df_gold_dim.write.format("delta").mode("append").save(dim_path)
logger.info(f"[LOAD] Dimension SCD Type 2 MERGE completed: {dim_path}")
```

## 6B. FACT TABLE PATTERN (Delta MERGE on grain key)

```python
# Read dimension tables for surrogate key resolution
def _read_dim(name, key_cols):
    path = f"{lakehouse_path}/Tables/{gold_schema}/dim_{name}"
    return spark.read.format("delta").load(path).filter("is_active = true").select(*key_cols)

df_dim_customer = _read_dim("customer", ["customer_id", "customer_sk"])
df_dim_product  = _read_dim("product",  ["product_id",  "product_sk"])

# Join Silver to Dimensions (always LEFT to preserve grain)
df_fact = df_silver \
    .join(broadcast(df_dim_customer), "customer_id", "left") \
    .join(broadcast(df_dim_product),  "product_id",  "left")

# Referential integrity check
null_sk = df_fact.filter(F.col("customer_sk").isNull()).count()
if null_sk > 0:
    logger.warning(f"[INTEGRITY] {null_sk} fact rows with NULL customer_sk (orphaned)")

# Calculate business measures from config
df_fact = df_fact \
    .withColumn("gross_amount",   F.col("quantity") * F.col("unit_price")) \
    .withColumn("discount_amount",F.col("gross_amount") * F.lit(discount_rate)) \
    .withColumn("taxable_amount", F.col("gross_amount") - F.col("discount_amount")) \
    .withColumn("tax_amount",     F.col("taxable_amount") * F.lit(tax_rate)) \
    .withColumn("net_amount",     F.col("taxable_amount") + F.col("tax_amount")) \
    .withColumn("fiscal_year",    F.year(F.col("order_date"))) \
    .withColumn("fiscal_quarter", F.quarter(F.col("order_date"))) \
    .withColumn("fiscal_month",   F.month(F.col("order_date"))) \
    .withColumn("fact_created_at",F.current_timestamp())

# Gold Fact Delta path
fact_path = f"{lakehouse_path}/Tables/{gold_schema}/fact_{config['entity_name']}"

if not DeltaTable.isDeltaTable(spark, fact_path):
    df_fact.limit(0).write.format("delta").mode("append").save(fact_path)

grain_merge  = " AND ".join([f"target.{k} = source.{k}" for k in grain_keys])
fact_delta   = DeltaTable.forPath(spark, fact_path)

fact_delta.alias("target").merge(
    df_fact.alias("source"),
    grain_merge
).whenMatchedUpdateAll(
).whenNotMatchedInsertAll(
).execute()

logger.info(f"[LOAD] Fact MERGE completed: {fact_path}")
```

## 6C. AGGREGATE / METRIC TABLE PATTERN (Full Refresh via Delta overwrite)

```python
if not aggregation_grain:
    raise ValueError("config['aggregation_grain'] must be provided for AGGREGATE table type")

df_agg = df_silver.groupBy(*aggregation_grain).agg(
    F.count("*").alias("record_count"),
    F.sum(config.get('measure_col', 'amount')).alias("total_amount"),
    F.avg(config.get('measure_col', 'amount')).alias("avg_amount"),
    F.min(config.get('date_col', 'order_date')).alias("first_date"),
    F.max(config.get('date_col', 'order_date')).alias("last_date")
)

# Revenue ranking window
from pyspark.sql.window import Window
period_cols = [g for g in aggregation_grain if "year" in g.lower() or "month" in g.lower()]
if period_cols:
    rank_win = Window.partitionBy(*period_cols).orderBy(F.desc("total_amount"))
    df_agg = df_agg \
        .withColumn("revenue_rank",       F.rank().over(rank_win)) \
        .withColumn("revenue_percentile", F.percent_rank().over(rank_win))

df_agg = df_agg.withColumn("aggregate_created_at", F.current_timestamp())

agg_path = f"{lakehouse_path}/Tables/{gold_schema}/agg_{config['metric_name']}"

# Aggregates use full overwrite (recalculated each run — deterministic)
df_agg.repartition(write_partitions).write \
    .format("delta") \
    .mode("overwrite") \
    .option("overwriteSchema", "true") \
    .save(agg_path)

logger.info(f"[LOAD] Aggregate table refreshed: {agg_path}")
```

### 8. POWER BI DIRECT LAKE COMPATIBILITY
Gold layer tables in Microsoft Fabric are consumed directly by Power BI via Direct Lake mode. Follow these rules:

```python
# Column naming: snake_case, no special chars, no spaces
# ✅ CORRECT
df = df.withColumnRenamed("ContactName",   "contact_name") \
       .withColumnRenamed("OrderID",       "order_id")

# ❌ FORBIDDEN
# df.withColumnRenamed("Contact Name", "Contact Name")  # Spaces cause Direct Lake issues

# Type placement rules for Power BI:
# - Numeric measures: DecimalType(18,4) — Fabric sums these correctly
# - Date keys: DateType — Power BI date hierarchy activated automatically
# - Boolean flags: BooleanType — renders as Yes/No slicer in PBI
# - Surrogate keys: LongType (not StringType) — faster joins in PBI data model
```
