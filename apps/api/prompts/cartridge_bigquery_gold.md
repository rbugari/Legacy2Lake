### 2. BIGQUERY GOLD LAYER FUNDAMENTALS

**⚠️ CRITICAL: Pure BigQuery SQL Scripting — no Spark, no Python.**

Gold has 3 structural patterns:
- **DIMENSION**: `TO_HEX(MD5(...))` hash SK + SCD Type 2 MERGE.
- **FACT**: INSERT INTO ... SELECT with dim JOINs for SK resolution + measure computation.
- **AGGREGATE**: `CREATE OR REPLACE TABLE ... AS SELECT` (CTAS — atomic, full overwrite); or `MATERIALIZED VIEW` for auto-refresh.

**BigQuery Gold advantages:**
- `CREATE OR REPLACE TABLE ... AS SELECT` is atomic, fastest BQ pattern for aggregates.
- `MATERIALIZED VIEW` auto-refreshes on source changes — zero orchestration cost for reporting.
- `BI Engine` acceleration: queries hitting Gold tables with `CLUSTER BY` columns can use in-memory BQ BI Engine.
- `AUTHORIZED VIEW` or `AUTHORIZED DATASET`: expose Gold to Looker/Data Studio without granting base table access.
- `TO_HEX(MD5(...))` and `FARM_FINGERPRINT(...)` are BigQuery-native hash functions for surrogate keys.

**Auto-dispatch from config:**
```
p_model_type = 'DIMENSION' | 'FACT' | 'AGGREGATE'
```

### 4. SURROGATE KEY — BIGQUERY HASH FUNCTIONS
```sql
-- Option A: MD5 hex (128-bit, URL-safe string, most common for BQ dim keys)
TO_HEX(MD5(CONCAT(
    COALESCE(CAST(BUSINESS_KEY_1 AS STRING), 'NULL'), '||',
    COALESCE(CAST(BUSINESS_KEY_2 AS STRING), 'NULL')
)))                 AS DIMENSION_SK

-- Option B: FARM_FINGERPRINT (64-bit INT64 — smaller, faster joins)
CAST(ABS(FARM_FINGERPRINT(CONCAT(
    COALESCE(CAST(BUSINESS_KEY_1 AS STRING), 'NULL'), '||',
    COALESCE(CAST(BUSINESS_KEY_2 AS STRING), 'NULL')
))) AS INT64)        AS DIMENSION_SK_INT
```

### 5. DIMENSION PATTERN (Hash SK + SCD Type 2 MERGE)
```sql
IF p_model_type = 'DIMENSION' THEN

    DECLARE v_silver_full STRING DEFAULT CONCAT('`', p_project, '.', p_silver_dataset, '.', p_source_table, '`');
    DECLARE v_gold_full   STRING DEFAULT CONCAT('`', p_project, '.', p_gold_dataset,   '.', p_target_table, '`');
    DECLARE v_incoming    STRING DEFAULT CONCAT('`', p_project, '.', p_gold_dataset,   '.INCOMING_', p_target_table, '`');

    -- Step 1: Build incoming with hash SK
    EXECUTE IMMEDIATE FORMAT("""
        CREATE OR REPLACE TEMP TABLE INCOMING_DIM AS
        SELECT
            TO_HEX(MD5(CONCAT(
                COALESCE(CAST(BUSINESS_KEY_1 AS STRING), 'NULL'), '||',
                COALESCE(CAST(BUSINESS_KEY_2 AS STRING), 'NULL')
            )))                          AS %s,
            BUSINESS_KEY_1,
            BUSINESS_KEY_2,
            ATTRIBUTE_A,
            CURRENT_TIMESTAMP()          AS VALID_FROM,
            CAST(NULL AS TIMESTAMP)      AS VALID_UNTIL,
            TRUE                         AS IS_CURRENT,
            CURRENT_TIMESTAMP()          AS GOLD_CREATED_AT,
            CURRENT_TIMESTAMP()          AS GOLD_UPDATED_AT
        FROM %s
        WHERE IS_CURRENT = TRUE
    """, p_sk_column, v_silver_full);

    -- Step 2: Create Gold dimension table if not exists
    DECLARE v_table_exists BOOL DEFAULT (
        SELECT COUNT(*) > 0
        FROM `region-us`.INFORMATION_SCHEMA.TABLES
        WHERE table_catalog = p_project
          AND table_schema  = p_gold_dataset
          AND table_name    = p_target_table
    );

    IF NOT v_table_exists THEN
        EXECUTE IMMEDIATE FORMAT("""
            CREATE TABLE %s (
                %s              STRING    NOT NULL,
                BUSINESS_KEY_1  STRING    NOT NULL,
                BUSINESS_KEY_2  STRING,
                ATTRIBUTE_A     STRING,
                VALID_FROM      TIMESTAMP NOT NULL,
                VALID_UNTIL     TIMESTAMP,
                IS_CURRENT      BOOL      NOT NULL,
                GOLD_CREATED_AT TIMESTAMP,
                GOLD_UPDATED_AT TIMESTAMP
            )
            PARTITION BY DATE(VALID_FROM)
            CLUSTER BY BUSINESS_KEY_1, IS_CURRENT
            OPTIONS (description = 'L2L Gold Dimension: {{asset_name}}')
        """, v_gold_full, p_sk_column);
    END IF;

    -- Step 3a: Expire changed records
    EXECUTE IMMEDIATE FORMAT("""
        MERGE %s T
        USING (SELECT * FROM INCOMING_DIM) S
        ON T.BUSINESS_KEY_1 = S.BUSINESS_KEY_1
           AND T.BUSINESS_KEY_2 = S.BUSINESS_KEY_2
           AND T.IS_CURRENT = TRUE
        WHEN MATCHED AND (
            T.ATTRIBUTE_A <> S.ATTRIBUTE_A
        ) THEN UPDATE SET
            T.VALID_UNTIL      = CURRENT_TIMESTAMP(),
            T.IS_CURRENT       = FALSE,
            T.GOLD_UPDATED_AT  = CURRENT_TIMESTAMP()
    """, v_gold_full);

    -- Step 3b: Insert new versions
    EXECUTE IMMEDIATE FORMAT("""
        INSERT INTO %s SELECT * FROM INCOMING_DIM
    """, v_gold_full);

END IF;
```

### 6. FACT PATTERN (SK Resolution + INSERT)
```sql
IF p_model_type = 'FACT' THEN

    DECLARE v_silver_fact STRING DEFAULT CONCAT('`', p_project, '.', p_silver_dataset, '.', p_source_table, '`');
    DECLARE v_gold_fact   STRING DEFAULT CONCAT('`', p_project, '.', p_gold_dataset,   '.', p_target_table, '`');

    EXECUTE IMMEDIATE FORMAT("""
        INSERT INTO %s
        SELECT
            -- SK lookups from Gold dimensions
            d1.DIMENSION1_SK,
            d2.DIMENSION2_SK,
            -- Degenerate dimensions
            f.TRANSACTION_CODE,
            f.ORDER_DATE,
            -- Partitioning date (for PARTITION BY)
            DATE(f.ORDER_DATE)              AS FACT_DATE,
            -- Business measures
            f.QUANTITY,
            f.UNIT_PRICE,
            ROUND(f.QUANTITY * f.UNIT_PRICE, 4)                             AS GROSS_AMOUNT,
            COALESCE(f.DISCOUNT, 0)                                         AS DISCOUNT_AMOUNT,
            ROUND(f.QUANTITY * f.UNIT_PRICE * (1 - COALESCE(f.DISCOUNT_PCT, 0)), 4) AS NET_AMOUNT,
            -- Audit
            CURRENT_TIMESTAMP()             AS GOLD_CREATED_AT
        FROM %s f
        LEFT JOIN `%s.%s.DIM_DIMENSION1` d1
            ON f.DIMENSION1_KEY = d1.BUSINESS_KEY_1 AND d1.IS_CURRENT = TRUE
        LEFT JOIN `%s.%s.DIM_DIMENSION2` d2
            ON f.DIMENSION2_KEY = d2.BUSINESS_KEY_1 AND d2.IS_CURRENT = TRUE
        WHERE f.IS_CURRENT = TRUE
          AND DATE(f.SILVER_CREATED_AT) >= DATE('%s')
    """,
    v_gold_fact,
    v_silver_fact,
    p_project, p_gold_dataset,   -- DIM_DIMENSION1 ref
    p_project, p_gold_dataset,   -- DIM_DIMENSION2 ref
    CAST(p_batch_date AS STRING)
    );

END IF;
```

### 7. AGGREGATE PATTERN — CTAS or MATERIALIZED VIEW

**Option A: CTAS — Full Overwrite (standard, orchestrator-driven):**
```sql
IF p_model_type = 'AGGREGATE' THEN

    EXECUTE IMMEDIATE FORMAT("""
        CREATE OR REPLACE TABLE `%s.%s.%s`
        PARTITION BY PERIOD_MONTH
        CLUSTER BY GROUP_DIM_1
        OPTIONS (description = 'L2L Gold Aggregate: {{asset_name}}')
        AS
        WITH base AS (
            SELECT *
            FROM `%s.%s.%s`
            WHERE IS_CURRENT = TRUE
        )
        SELECT
            GROUP_DIM_1,
            GROUP_DIM_2,
            DATE_TRUNC(TRANSACTION_DATE, MONTH)         AS PERIOD_MONTH,
            COUNT(*)                                    AS RECORD_COUNT,
            APPROX_COUNT_DISTINCT(BUSINESS_KEY_1)       AS UNIQUE_ENTITIES,
            SUM(AMOUNT)                                 AS TOTAL_AMOUNT,
            AVG(AMOUNT)                                 AS AVG_AMOUNT,
            MAX(AMOUNT)                                 AS MAX_AMOUNT,
            MIN(AMOUNT)                                 AS MIN_AMOUNT,
            SUM(SUM(AMOUNT)) OVER (
                PARTITION BY GROUP_DIM_1
                ORDER BY DATE_TRUNC(TRANSACTION_DATE, MONTH)
                ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
            )                                           AS RUNNING_TOTAL,
            CURRENT_TIMESTAMP()                         AS GOLD_CREATED_AT,
            CURRENT_DATE()                              AS AGGREGATE_DATE
        FROM base
        GROUP BY 1, 2, 3
    """,
    p_project, p_gold_dataset, p_target_table,
    p_project, p_silver_dataset, p_source_table
    );

END IF;
```

**Option B: MATERIALIZED VIEW (auto-refresh, declarative — if config['use_mv'] = True):**
```sql
-- Generated as a standalone DDL (run once by orchestrator):
-- BigQuery auto-refreshes MVs when source data changes (no Airflow needed).
CREATE MATERIALIZED VIEW `{{project}}.{{gold_dataset}}.{{asset_name}}_AGG`
OPTIONS (
    enable_refresh = true,
    refresh_interval_minutes = 60
)
AS
SELECT
    GROUP_DIM_1,
    DATE_TRUNC(TRANSACTION_DATE, MONTH)     AS PERIOD_MONTH,
    COUNT(*)                                AS RECORD_COUNT,
    SUM(AMOUNT)                             AS TOTAL_AMOUNT
FROM `{{project}}.{{silver_dataset}}.{{source_table}}`
WHERE IS_CURRENT = TRUE
GROUP BY 1, 2;
```

**Materialized Views — Auto-Refresh Gold Aggregates:**
```sql

CREATE MATERIALIZED VIEW `{{project}}.{{gold_dataset}}.MV_{{target_table}}_AGG`
OPTIONS (
    enable_refresh                    = true,
    refresh_interval_minutes          = 60,
    allow_non_incremental_definition  = true,
    labels = [("l2l_layer", "gold"), ("l2l_asset", "{{asset_name}}")]
)
AS
SELECT
    GROUP_DIM_1,
    DATE_TRUNC(TRANSACTION_DATE, MONTH)     AS PERIOD_MONTH,
    COUNT(*)                                AS RECORD_COUNT,
    SUM(AMOUNT)                             AS TOTAL_AMOUNT
FROM `{{project}}.{{silver_dataset}}.{{source_table}}`
WHERE IS_CURRENT = TRUE
GROUP BY 1, 2;
-- BQ auto-refreshes every 60 min. No Airflow/Cloud Composer scheduling needed.
```

**Result Cache — Gold Aggregate Queries:**
```sql
-- BQ Result Cache (24h TTL) applies to Gold when:
-- ✅ No CURRENT_TIMESTAMP(), CURRENT_DATE(), NOW() in SELECT list
```

```sql

    job_id,
```
