### 2. SNOWFLAKE SQL GOLD FUNDAMENTALS

**⚠️ CRITICAL: This cartridge generates SQL Stored Procedures, NOT Python/Snowpark code.**

Gold layer has **3 structural patterns**, each mapping to a different SQL approach:
- **DIMENSION**: `MERGE INTO` with SHA2-based surrogate key + SCD Type 2 historical tracking.
- **FACT**: `INSERT INTO ... SELECT` joining Silver dims to resolve surrogate keys + compute measures.
- **AGGREGATE**: `CREATE OR REPLACE TABLE ... AS SELECT` (CTAS) — full overwrite each run. Or optionally a `DYNAMIC TABLE` for near-real-time refresh.

**Snowflake SQL Gold advantages:**
- `CREATE OR REPLACE TABLE ... AS SELECT` (CTAS) is fully atomic and the fastest aggregate pattern.
- `DYNAMIC TABLE` replaces scheduled CTAS with declarative lag-based refresh — zero orchestration overhead.
- `SHA2(CONCAT_WS('||', ...)` generates deterministic surrogate keys natively in SQL.
- `QUALIFY` eliminates subquery-heavy deduplication.
- `CLUSTERING KEYS` defined at DDL level — no per-run maintenance.

### 4. DIMENSION PATTERN (SHA2 SK + SCD Type 2 MERGE)

```sql
-- Gold Dimension: deterministic SHA2 surrogate key + SCD Type 2
IF (P_MODEL_TYPE = 'DIMENSION') THEN

    LET v_silver_full STRING DEFAULT P_DATABASE || '.' || P_SILVER_SCHEMA || '.' || P_SOURCE_TABLE;
    LET v_gold_full   STRING DEFAULT P_DATABASE || '.' || P_GOLD_SCHEMA   || '.' || P_TARGET_TABLE;

    -- Step 1: Build incoming dimension with SK
    EXECUTE IMMEDIATE '
        CREATE OR REPLACE TRANSIENT TABLE ' || P_DATABASE || '.' || P_GOLD_SCHEMA || '.INCOMING_' || P_TARGET_TABLE || ' AS
        SELECT
            SHA2(
                CONCAT_WS(''||'',
                    COALESCE(BUSINESS_KEY_1::STRING, ''NULL''),
                    COALESCE(BUSINESS_KEY_2::STRING, ''NULL'')
                ), 256
            )                       AS ' || P_SK_COLUMN || ',
            BUSINESS_KEY_1,
            BUSINESS_KEY_2,
            ATTRIBUTE_A,
            CURRENT_TIMESTAMP()     AS VALID_FROM,
            NULL::TIMESTAMP_NTZ     AS VALID_UNTIL,
            TRUE                    AS IS_CURRENT,
            CURRENT_TIMESTAMP()     AS GOLD_CREATED_AT,
            CURRENT_TIMESTAMP()     AS GOLD_UPDATED_AT
        FROM ' || v_silver_full || '
        WHERE IS_CURRENT = TRUE
    ';

    -- Step 2: Create Gold dimension table if not exists
    EXECUTE IMMEDIATE '
        CREATE TABLE IF NOT EXISTS ' || v_gold_full || ' (
            ' || P_SK_COLUMN || '   STRING        NOT NULL,
            BUSINESS_KEY_1          STRING        NOT NULL,
            BUSINESS_KEY_2          STRING,
            ATTRIBUTE_A             STRING,
            VALID_FROM              TIMESTAMP_NTZ NOT NULL,
            VALID_UNTIL             TIMESTAMP_NTZ,
            IS_CURRENT              BOOLEAN       NOT NULL,
            GOLD_CREATED_AT         TIMESTAMP_NTZ,
            GOLD_UPDATED_AT         TIMESTAMP_NTZ
        )
        CLUSTER BY (BUSINESS_KEY_1, IS_CURRENT)
        COMMENT = ''L2L Gold Dimension SCD2: {{asset_name}}''
    ';

    LET v_incoming STRING DEFAULT P_DATABASE || '.' || P_GOLD_SCHEMA || '.INCOMING_' || P_TARGET_TABLE;

    -- Step 3a: Expire changed records
    EXECUTE IMMEDIATE '
        MERGE INTO ' || v_gold_full || ' t
        USING ' || v_incoming || ' s
        ON t.BUSINESS_KEY_1 = s.BUSINESS_KEY_1
           AND t.BUSINESS_KEY_2 = s.BUSINESS_KEY_2
           AND t.IS_CURRENT = TRUE
        WHEN MATCHED AND (
            t.ATTRIBUTE_A <> s.ATTRIBUTE_A
            -- extend with all trackable attributes
        ) THEN UPDATE SET
            t.VALID_UNTIL      = CURRENT_TIMESTAMP(),
            t.IS_CURRENT       = FALSE,
            t.GOLD_UPDATED_AT  = CURRENT_TIMESTAMP()
    ';

    -- Step 3b: Insert new versions
    EXECUTE IMMEDIATE '
        INSERT INTO ' || v_gold_full || '
        SELECT * FROM ' || v_incoming || '
    ';

    EXECUTE IMMEDIATE 'DROP TABLE IF EXISTS ' || v_incoming;

END IF;
```

### 5. FACT PATTERN (SK Resolution + INSERT INTO)

```sql
IF (P_MODEL_TYPE = 'FACT') THEN

    LET v_silver_fact STRING DEFAULT P_DATABASE || '.' || P_SILVER_SCHEMA || '.' || P_SOURCE_TABLE;
    LET v_gold_fact   STRING DEFAULT P_DATABASE || '.' || P_GOLD_SCHEMA   || '.' || P_TARGET_TABLE;

    -- Resolve surrogate keys from Gold dimensions + compute measures
    -- Dimension reference tables list comes from P_BUSINESS_KEYS
    -- (extended config, see notes below)

    EXECUTE IMMEDIATE '
        INSERT INTO ' || v_gold_fact || '
        SELECT
            -- Surrogate key lookups (one per dimension)
            d1.' || P_SK_COLUMN || '        AS DIMENSION1_SK,
            d2.DIMENSION2_SK                AS DIMENSION2_SK,
            -- Degenerate dimensions (from fact source directly)
            f.TRANSACTION_CODE,
            f.ORDER_DATE,
            -- Business measures
            f.QUANTITY,
            f.UNIT_PRICE,
            ROUND(f.QUANTITY * f.UNIT_PRICE, 4) AS GROSS_AMOUNT,
            COALESCE(f.DISCOUNT, 0)              AS DISCOUNT_AMOUNT,
            ROUND(f.QUANTITY * f.UNIT_PRICE * (1 - COALESCE(f.DISCOUNT_PCT, 0)), 4) AS NET_AMOUNT,
            -- Audit
            CURRENT_TIMESTAMP()             AS GOLD_CREATED_AT
        FROM ' || v_silver_fact || ' f
        LEFT JOIN ' || P_DATABASE || '.' || P_GOLD_SCHEMA || '.DIM_DIMENSION1 d1
            ON f.DIMENSION1_KEY = d1.BUSINESS_KEY_1 AND d1.IS_CURRENT = TRUE
        LEFT JOIN ' || P_DATABASE || '.' || P_GOLD_SCHEMA || '.DIM_DIMENSION2 d2
            ON f.DIMENSION2_KEY = d2.BUSINESS_KEY_1 AND d2.IS_CURRENT = TRUE
        WHERE f.IS_CURRENT = TRUE
          AND f.SILVER_CREATED_AT >= ''' || P_BATCH_DATE::STRING || '''
    ';

END IF;
```

### 6. AGGREGATE PATTERN (CTAS Full Overwrite or DYNAMIC TABLE)

**Option A: CTAS — Full overwrite (standard):**
```sql
IF (P_MODEL_TYPE = 'AGGREGATE') THEN

    LET v_silver_src STRING DEFAULT P_DATABASE || '.' || P_SILVER_SCHEMA || '.' || P_SOURCE_TABLE;
    LET v_gold_agg   STRING DEFAULT P_DATABASE || '.' || P_GOLD_SCHEMA   || '.' || P_TARGET_TABLE;

    EXECUTE IMMEDIATE '
        CREATE OR REPLACE TABLE ' || v_gold_agg || '
        CLUSTER BY (GROUP_DIM_1, PERIOD_MONTH)
        COMMENT = ''L2L Gold Aggregate: {{asset_name}}''
        AS
        WITH base AS (
            SELECT *
            FROM ' || v_silver_src || '
            WHERE IS_CURRENT = TRUE
        )
        SELECT
            GROUP_DIM_1,
            GROUP_DIM_2,
            DATE_TRUNC(''MONTH'', TRANSACTION_DATE)     AS PERIOD_MONTH,
            COUNT(*)                                    AS RECORD_COUNT,
            COUNT(DISTINCT BUSINESS_KEY_1)              AS UNIQUE_ENTITIES,
            SUM(AMOUNT)                                 AS TOTAL_AMOUNT,
            AVG(AMOUNT)                                 AS AVG_AMOUNT,
            MAX(AMOUNT)                                 AS MAX_AMOUNT,
            MIN(AMOUNT)                                 AS MIN_AMOUNT,
            -- Running total (window function in CTAS is legal in Snowflake)
            SUM(SUM(AMOUNT)) OVER (
                PARTITION BY GROUP_DIM_1
                ORDER BY DATE_TRUNC(''MONTH'', TRANSACTION_DATE)
                ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
            )                                           AS RUNNING_TOTAL,
            CURRENT_TIMESTAMP()                         AS GOLD_CREATED_AT,
            CURRENT_DATE()                              AS AGGREGATE_DATE
        FROM base
        GROUP BY 1, 2, 3
    ';

END IF;
```

**Option B: DYNAMIC TABLE (near-real-time, declare once — orchestrator runs ONCE at setup):**
```sql
-- ⚠️ DYNAMIC TABLE: Generated as a DDL block, not inside a SP call.
-- Orchestrator creates it ONCE. Snowflake refreshes automatically.
-- Only generate this if config['use_dynamic_table'] = True

CREATE OR REPLACE DYNAMIC TABLE {{schema}}.{{ASSET_NAME}}_GOLD_AGG
    TARGET_LAG = '{{dynamic_table_lag}}'        -- e.g. '1 hour', '5 minutes'
    WAREHOUSE  = {{warehouse_name}}             -- comes from config (NEVER hardcode)
    COMMENT    = 'L2L Gold Aggregate Dynamic Table: {{asset_name}}'
AS
SELECT
    GROUP_DIM_1,
    DATE_TRUNC('MONTH', TRANSACTION_DATE)   AS PERIOD_MONTH,
    COUNT(*)                                AS RECORD_COUNT,
    SUM(AMOUNT)                             AS TOTAL_AMOUNT,
    CURRENT_TIMESTAMP()                     AS LAST_REFRESHED
FROM {{schema}}.{{SOURCE_SILVER_TABLE}}
WHERE IS_CURRENT = TRUE
GROUP BY 1, 2;
```

### 9. SNOWFLAKE NATIVE OPTIMIZATION HINTS

**QUERY_TAG — Cost Attribution (MANDATORY):**

**DYNAMIC TABLES — Declarative Incremental Materialization (Gold Alternative):**
```sql
-- Drop-in replacement for the AGGREGATE CTAS pattern.
-- Snowflake computes incremental delta automatically — no scheduled SP needed.
-- Use when Gold aggregate staleness of 1h or more is acceptable.
CREATE OR REPLACE DYNAMIC TABLE {{database}}.{{gold_schema}}.{{target_table}}_DYN
    TARGET_LAG = '1 hour'           -- acceptable staleness: '15 minutes', '1 day', etc.
    WAREHOUSE  = {{warehouse_name}} -- Dynamic Table uses its own refresh credits
    AS
    SELECT
        GROUP_DIM_1,
        DATE_TRUNC('MONTH', TRANSACTION_DATE)   AS PERIOD_MONTH,
        COUNT(*)                                AS RECORD_COUNT,
        SUM(AMOUNT)                             AS TOTAL_AMOUNT,
        AVG(AMOUNT)                             AS AVG_AMOUNT
    FROM {{database}}.{{silver_schema}}.{{source_table}}
    WHERE IS_CURRENT = TRUE
    GROUP BY 1, 2;

-- Force immediate refresh before a downstream report:
ALTER DYNAMIC TABLE {{database}}.{{gold_schema}}.{{target_table}}_DYN REFRESH;

-- Monitor lag and status:
SHOW DYNAMIC TABLES LIKE '{{target_table}}_DYN' IN DATABASE {{database}};
```

**Result Cache — Gold Aggregate Queries:**
```sql
-- Gold aggregates are prime candidates for Snowflake Result Cache (24h TTL).
-- RULES to keep Gold results cacheable:
-- ✅ No CURRENT_TIMESTAMP() / CURRENT_DATE() in SELECT list
-- ✅ No non-deterministic UDFs in SELECT
-- ✅ Source table data unchanged between queries
-- BAD  (breaks cache): SELECT CURRENT_TIMESTAMP() AS REPORT_TS, SUM(AMOUNT) FROM gold.agg
-- GOOD (cacheable):    SELECT PERIOD_MONTH, SUM(AMOUNT) FROM gold.agg
-- Pass volatile values as SP parameters — keep the SELECT clause deterministic.
```

**CLUSTER BY for Gold Dimensions and Facts:**
```sql
-- Dimensions: cluster by primary lookup pattern
ALTER TABLE {{database}}.{{gold_schema}}.DIM_{{entity}}
    CLUSTER BY (BUSINESS_KEY_1, IS_CURRENT);

-- Facts: cluster by temporal partition + most selective dimension key
ALTER TABLE {{database}}.{{gold_schema}}.FACT_{{entity}}
    CLUSTER BY (FACT_DATE::DATE, DIMENSION_SK);
```
